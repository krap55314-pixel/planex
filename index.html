PESOS = {

# Rangos de normalización

def diferencia_normalizada(valor1, valor2, rango):
    diferencia = abs(valor1 - valor2)
    return min(1.0, diferencia / rango)


def calcular_similitud(region, lugar, pesos=None):
    if pesos is None:
        pesos = PESOS
    penalizacion = 0

    # CICLO FOR: recorrer variables numéricas
    for variable in RANGOS:
        diferencia = diferencia_normalizada(
            region[variable],
            lugar[variable],
            RANGOS[variable]
        )

        penalizacion += pesos[variable] * diferencia

    # Comparación geológica simplificada
    if region["geologia"] == lugar["geologia"]:
        diferencia_geologica = 0
    else:
        diferencia_geologica = 1

    penalizacion += (
        pesos["geologia"] * diferencia_geologica
    )

    similitud = 100 * (1 - penalizacion)

    return round(max(0, min(100, similitud)), 2)


def buscar_analogos(region, pesos=None):
    resultados = []

    # CICLO FOR: analizar cada lugar terrestre
    for lugar in LUGARES_TIERRA:

        puntaje = calcular_similitud(region, lugar, pesos)

        resultados.append({
            "nombre": lugar["nombre"],
            "similitud": puntaje
        })

    # Ordenar del más parecido al menos parecido
    resultados.sort(
        key=lambda resultado: resultado["similitud"],
        reverse=True
    )

    return resultados[:5]

def normalizar_pesos(pesos):
    """Hace que la suma de pesos sea 1. Si todos son 0, usa pesos iguales."""
    total = sum(max(0.0, float(pesos.get(v, 0))) for v in PESOS)
    if total <= 0:
        return {v: 1 / len(PESOS) for v in PESOS}
    return {v: max(0.0, float(pesos.get(v, 0))) / total for v in PESOS}


def explicar_similitud(region, lugar, pesos=None):
    """Devuelve el detalle por variable: d (diferencia normalizada), peso y aporte w*d."""
    if pesos is None:
        pesos = PESOS
    detalle = []
    for variable in RANGOS:
        d = diferencia_normalizada(region[variable], lugar[variable], RANGOS[variable])
        detalle.append({"variable": variable, "valor_region": region[variable],
                        "valor_lugar": lugar[variable], "d": d,
                        "peso": pesos[variable], "aporte": pesos[variable] * d})
    d = 0 if region["geologia"] == lugar["geologia"] else 1
    detalle.append({"variable": "geologia", "valor_region": region["geologia"],
                    "valor_lugar": lugar["geologia"], "d": d,
                    "peso": pesos["geologia"], "aporte": pesos["geologia"] * d})
    return detalle


def analizar(cuerpo, nombre_region, pesos=None):
    """Ejecuta el motor para una región y devuelve el Top 5 con su detalle matemático."""
    region = next((r for r in REGIONES[cuerpo] if r["nombre"] == nombre_region), None)
    if region is None:
        raise KeyError(nombre_region)
    pn = normalizar_pesos(pesos) if pesos is not None else dict(PESOS)
    lugares = {l["nombre"]: l for l in LUGARES_TIERRA}
    resultados = []
    for r in buscar_analogos(region, pn):
        resultados.append({**r, "detalle": explicar_similitud(region, lugares[r["nombre"]], pn)})
    return {"cuerpo": cuerpo, "region": region, "pesos_normalizados": pn, "resultados": resultados}

Detalle matemático

Cómo se calcula

S = 100 × (1 − Σ wᵢ · dᵢ)
  • wᵢ: peso de la variable i, normalizado para que Σ wᵢ = 1.
  • dᵢ numérica = min(1, |Marte − Tierra| / rango de normalización).
  • dᵢ geológica = 1 − afinidad entre tipos de terreno (matriz inventada).
  • Con Σ wᵢ = 1 y dᵢ ∈ [0,1], S queda siempre entre 0 y 100.
Rangos usados: temperatura 120 °C, pendiente 30°, elevación 10 000 m, aridez e hielo 1.0.

Variables terrestres

  • Temperatura: media de la temperatura diaria a 2 m, ERA5, 2015–2024.
  • Aridez: 1 − IA/0,65, con IA = precipitación / ET0 FAO (índice de aridez UNEP; 0,65 es el umbral de clima húmedo).
  • Hielo/permafrost: (−1 − T media)/7, limitado a 0–1. Es un proxy: el permafrost suele aparecer con medias anuales bajo −1 °C y es continuo bajo −8 °C.
  • Elevación y pendiente: Copernicus DEM GLO-90; pendiente media en una grilla 7×7 de ~9 km alrededor del sitio.

Fuentes de datos

  • ERA5 (Copernicus Climate Change Service / ECMWF), servido por la API histórica de Open-Meteo: temperatura, precipitación y ET0 diarias 2015–2024.
  • Copernicus DEM GLO-90, vía la API de elevación de Open-Meteo: elevación y pendiente.
  • Natural Earth 1:110M: contornos continentales del mapa.
  • Globos 3D: mosaico color Viking MDIM 2.1 y altimetría MOLA (NASA/USGS, vía Mars Trek); mosaico LRO WAC y altimetría LOLA (NASA, vía Moon Trek); Blue Marble y Earth at Night (NASA, vía three-globe). El relieve está exagerado para que se lea a escala planetaria.
  • Marte: altimetría MOLA (Mars Global Surveyor) y temperaturas de las estaciones REMS (Curiosity) y MEDA (Perseverance), redondeadas.
  • Luna: temperaturas LRO Diviner y altimetría LOLA, redondeadas.
Los datos terrestres se descargaron el 6 de noviembre de 2026 y van incluidos en la página: no se consultan en vivo. Los valores de Marte y la Luna son referencias aproximadas, no mediciones puntuales; pendiente, aridez, hielo y la matriz de afinidad geológica son estimaciones del modelo.

import numpy as np
import pandas as pd

RANGES = {"temp": 120, "slope": 30, "elev": 10000, "arid": 1, "ice": 1}

def similarity(mars: dict, earth: pd.DataFrame, weights: dict, geo_sim: dict):
    w = pd.Series(weights, dtype=float)
    w = w / w.sum()                      # Σ wᵢ = 1
    d = pd.DataFrame(index=earth.index)
    for k, rng in RANGES.items():
        d[k] = (earth[k] - mars[k]).abs().div(rng).clip(upper=1)
    d["geo"] = earth["geo"].map(lambda g: 1 - geo_sim[mars["geo"]][g])
    s = 100 * (1 - (d[w.index] * w).sum(axis=1))
    return earth.assign(S=s).sort_values("S", ascending=False).head(5)
